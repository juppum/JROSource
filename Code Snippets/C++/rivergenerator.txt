#include <cmath>
#include <vector>

class RiverGenerator {
public:
    float screenWidth = 800.0f;
    float baseRiverWidth = 300.0f; // How wide the "water" is
    float difficultyScaling = 0.05f; // Makes it narrower over time

    struct RiverBoundaries {
        float leftWall;
        float rightWall;
    };

    // This is the core "Seed" logic
    RiverBoundaries getBoundariesAt(float distance) {
        // Create a shifting center point using combined Sine waves
        float movement = std::sin(distance * 0.005f) * 100.0f;
        movement += std::sin(distance * 0.01f) * 40.0f; // Adds jaggedness
        
        float currentCenter = screenWidth / 2.0f + movement;
        
        // Narrow the river as the player goes further (Diablo difficulty scaling)
        float currentWidth = baseRiverWidth - (distance * difficultyScaling);
        if (currentWidth < 100.0f) currentWidth = 100.0f; // Minimum width

        return { currentCenter - (currentWidth / 2.0f), currentCenter + (currentWidth / 2.0f) };
    }
};
