struct LootItem {
    std::string name;
    Rarity rarity;
    sf::Color color;
    // Stats for the ARPG side
    float fuelBonus;
    float thrustBonus;
};

#include <random>

class LootGenerator {
public:
    LootItem rollForLoot() {
        int roll = rand() % 1000; // 0 to 999

        if (roll < 5)        return {"Ancient Turbine", Rarity::Legendary, sf::Color(255, 165, 0)}; // 0.5%
        if (roll < 50)       return {"Heavy Plating", Rarity::Rare, sf::Color::Yellow};            // 4.5%
        if (roll < 200)      return {"Tuned Engine", Rarity::Magic, sf::Color::Blue};             // 15%
        
        return {"Scrap Metal", Rarity::Common, sf::Color::White};                                 // 80%
    }
};

class DroppedItem {
public:
    sf::Vector2f position;
    sf::Vector2f velocity; // For the "burst" effect
    LootItem data;
    bool isCollected = false;

    void update(float dt, float scrollSpeed) {
        // Items drift down with the river
        position.y += (scrollSpeed * 0.5f) * dt; 
        // Apply some "burst" friction
        position += velocity * dt;
        velocity *= 0.95f; 
    }
};

void onBridgeDestroyed(sf::Vector2f bridgePos) {
    int dropCount = 5 + (rand() % 5); // 5 to 10 items explode out
    
    for (int i = 0; i < dropCount; i++) {
        DroppedItem item;
        item.data = lootGen.rollForLoot();
        item.position = bridgePos;
        
        // Random "explosion" direction
        float angle = (rand() % 360) * 3.14159f / 180.0f;
        float speed = 100.0f + (rand() % 200);
        item.velocity = sf::Vector2f(cos(angle) * speed, sin(angle) * speed);
        
        activeDrops.push_back(item);
    }
}
